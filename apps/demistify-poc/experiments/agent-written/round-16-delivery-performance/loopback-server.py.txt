from http.server import ThreadingHTTPServer,SimpleHTTPRequestHandler
from pathlib import Path
from urllib.parse import unquote,urlsplit
import hashlib,json,threading,time,sys
root=Path(sys.argv[1]).resolve();public=root/'public';lock=threading.Lock()
class Handler(SimpleHTTPRequestHandler):
 def do_GET(self):
  path=unquote(urlsplit(self.path).path);relative=path.lstrip('/');target=(public/relative).resolve()
  if not target.is_relative_to(public):self.send_error(403);return
  if target.is_dir():target=target/'index.html'
  if not target.is_file():self.send_error(404);return
  data=target.read_bytes();etag='"'+hashlib.sha256(data).hexdigest()+'"';code=304 if self.headers.get('If-None-Match')==etag else 200
  self.send_response(code);self.send_header('ETag',etag);self.send_header('Cache-Control','no-cache' if target.suffix=='.html' else 'public,max-age=31536000,immutable');self.send_header('Content-Type',self.guess_type(str(target)));self.send_header('Access-Control-Allow-Origin','*')
  if code==200:self.send_header('Content-Length',str(len(data)))
  self.end_headers()
  if code==200:self.wfile.write(data)
  with lock:
   with (root/'http-requests.jsonl').open('a') as output:output.write(json.dumps({'path':path,'status':code,'entityBytes':len(data) if code==200 else 0,'at':time.monotonic()})+'\n')
 def log_message(self,*args):pass
server=ThreadingHTTPServer(('127.0.0.1',4190),Handler)
print('Performance fixture listening on loopback4190',flush=True);server.serve_forever()
